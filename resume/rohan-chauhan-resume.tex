\section{Projects}
\project{Lumida --- AI Data Anomaly Detection Engine}{Self-hosted engine that connects read-only to company databases, finds hidden data problems, explains them with an LLM and learns from human feedback.}{lumida.netlify.app}
\begin{itemize}
  \item Infers undocumented relationships between tables and flags data problems, each with evidence and a plain-language LLM explanation.
  \item Human-in-the-loop: reviewers confirm or dismiss findings, and the engine learns from that feedback.
  \item On a test database it found \textbf{15/15} hidden relationships and \textbf{13/13} planted problems with \textbf{0} false alarms.
  \item \tech{Python, FastAPI, PostgreSQL, Docker, Google Gemini}
\end{itemize}
\vspace{4pt}

\project{Distiller --- LLM Cost-Reduction Gateway (in progress)}{OpenAI-compatible gateway that teams adopt by changing one line.}{}
\begin{itemize}
  \item Logs a company's repetitive LLM calls, fine-tunes small open models with LoRA to copy the large model, tests them in shadow mode, then routes confident requests to the cheaper model with fallback.
  \item Gateway skeleton and CLI working; training and routing in development.
  \item \tech{Python, FastAPI, LiteLLM, LoRA, Ollama}
\end{itemize}
\vspace{4pt}

\project{Bug Detection \& Fix Suggestion Tool}{VS Code extension for AI-assisted debugging.}{}

  \item Captures both static lint errors and runtime exceptions, then uses LLM-based RAG over a vector database to explain or fix them.
  \item Two modes: \textit{Tutor} explains why the code broke; \textit{Debugger} proposes a fix.
  \item \tech{TypeScript, VS Code API, FastAPI, Python, RAG, vector DB}
\end{itemize}
